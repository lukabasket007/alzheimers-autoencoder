"""

"""### Latent Space Visualization with UMAP

"""### Evaluating Latent Space Discriminability with a Classifier

To quantitatively assess how well the autoencoder's latent space differentiates between the two classes, we can train a simple classifier on the extracted latent features. If the latent space is discriminative, a simple classifier should achieve reasonable accuracy.
"""

from tensorflow.keras import layers, models
from sklearn.metrics import accuracy_score

# Always recreate the encoder model from the latest autoencoder
# This ensures it uses the most up-to-date autoencoder and its learned weights.
encoder = models.Model(inputs=autoencoder.input, outputs=encoded)

# 1. Extract latent features for training and test sets

# Check if X_train and X_test are not empty
if len(X_train) > 0 and len(X_test) > 0:
    # Debug prints to check X_train and X_test shapes right before use
    print(f"Debug: X_train shape in classifier cell: {X_train.shape}")
    print(f"Debug: X_test shape in classifier cell: {X_test.shape}")

    # Get latent space representations for the training set
    latent_train = encoder.predict(X_train)
    latent_train_flat = latent_train.reshape(latent_train.shape[0], -1)

    # Get latent space representations for the test set
    latent_test = encoder.predict(X_test)
    latent_test_flat = latent_test.reshape(latent_test.shape[0], -1)

    print(f"Shape of flattened latent train features: {latent_train_flat.shape}")
    print(f"Shape of flattened latent test features: {latent_test_flat.shape}")

    # Determine the number of unique classes for the classifier
    num_classes = len(np.unique(y_train)) # Use y_train to get all class labels

    # 2. Define a simple classifier model for multi-class classification
    latent_dim = latent_train_flat.shape[1]

    classifier_input = layers.Input(shape=(latent_dim,))
    x = layers.Dense(64, activation='relu')(classifier_input)
    x = layers.Dropout(0.3)(x)
    x = layers.Dense(32, activation='relu')(x)
    x = layers.Dropout(0.3)(x)
    # Changed to num_classes neurons with 'softmax' activation for multi-class classification
    classifier_output = layers.Dense(num_classes, activation='softmax')(x)

    classifier_model = models.Model(classifier_input, classifier_output)

    # Compile the classifier with sparse_categorical_crossentropy for integer labels
    classifier_model.compile(optimizer='adam', loss='sparse_categorical_crossentropy', metrics=['accuracy'])

    classifier_model.summary()

    # 3. Train the classifier
    print("\nTraining the classifier on latent features...")
    history_classifier = classifier_model.fit(latent_train_flat, y_train,
                                            epochs=20, # Can adjust epochs for the classifier
                                            batch_size=32,
                                            validation_data=(latent_test_flat, y_test))

    # 4. Evaluate the classifier
    print("\nEvaluating the classifier on the test set...")
    loss, accuracy = classifier_model.evaluate(latent_test_flat, y_test, verbose=0)
    print(f"Classifier Test Accuracy: {accuracy*100:.2f}%")

    # Plot training & validation accuracy values
    plt.figure(figsize=(10, 6))
    plt.plot(history_classifier.history['accuracy'], label='Classifier Training Accuracy')
    plt.plot(history_classifier.history['val_accuracy'], label='Classifier Validation Accuracy')
    plt.title('Classifier Accuracy During Training on Latent Features')
    plt.ylabel('Accuracy')
    plt.xlabel('Epoch')
    plt.legend(loc='lower right')
    plt.grid(True)
    plt.show()

else:
    print("Skipping classifier training: X_train or X_test is empty because no images were loaded.")
